-- FLIN bubble sort; returns integer list in increasing order
-- call as e.g. bsort([5,3,1])
-- function order is so always know number outputs when used in RHS

bubble([],x)    = M([x])
bubble(M(xs),x) = M(x:xs)
bubble(_y:ys,x) = ? x<y => x:bubble(ys,y) ? _ => y:bubble(ys,x)

bsort([])    = []
bsort(x:xs)  = bsort(bubble(xs,x))
bsort(M(xs)) = xs
